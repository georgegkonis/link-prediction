\begin{frame}{Συνεισφορές}
  \vspace{0.3cm}
  \begin{enumerate}
    \setlength{\itemsep}{0.55cm}
    \item \textbf{\en{CascadeLP}:} κλιμακωτός ταξινομητής που συνδυάζει δομή
      και κείμενο και στέλνει στα ακριβά επίπεδα μόνο τα αβέβαια ζεύγη:
      ανταγωνιστική ακρίβεια με κλάσμα του κόστους.
    \item \textbf{Διάγνωση του \en{DSAA 2023}:} εξήγηση των σχεδόν τέλειων
      βαθμολογιών του δημόσιου συνόλου μέσω στρέβλωσης αρνητικών και
      διαρροής γραφήματος.
    \item \textbf{Νέο σύνολο αξιολόγησης:} \WfCrawlNodes{} άρθρα και
      \WfCrawlEdges{} πραγματικοί υπερσύνδεσμοι από τα επίσημα
      \en{dumps} της \en{Wikipedia}, με αυστηρό πρωτόκολλο.
    \item \textbf{Έλεγχος σε αραιό γράφημα:} με λίγη δομή και δύσκολα
      αρνητικά, η δρομολόγηση ανά ζεύγος αποδίδει το μεγαλύτερο κέρδος.
  \end{enumerate}
\end{frame}

\begin{frame}{Πραγματικά γραφήματα είναι ελλιπή}
  \centering Κρατάμε μόνο το \textbf{\WsRetentionPct\%} των ακμών ως
  παρατηρούμενο γράφημα· το υπόλοιπο γίνεται θετικά ζεύγη ελέγχου.
  \vspace{0.4cm}

  \begin{columns}[c,onlytextwidth]
    \column{0.36\textwidth}
      \bignum{\WfMeanDegree{}\,→\,\WsObservedMeanDegree{}}{μέσος βαθμός κόμβου}
      \vspace{0.3cm}
      \bignum{\WsHeldOutEdges{}}{κρυμμένες πραγματικές ακμές}
    \column{0.60\textwidth}
      \begin{block}{Τυχαία αρνητικά}
        Ομοιόμορφα μη συνδεδεμένα ζεύγη: τα περισσότερα είναι εύκολα.
      \end{block}
      \vspace{0.15cm}
      \begin{block}{Δύσκολα αρνητικά}
        Μη συνδεδεμένα ζεύγη \textbf{με κοινό γείτονα}: μοιάζουν δομικά με

      \end{block}
  \end{columns}
  \vspace{0.3cm}
  \takeaway{Το πιο ρεαλιστικό σενάριο: λίγη δομή, και αρνητικά που «μοιάζουν»
    με συνδέσμους.}

\begin{frame}{Συμπεράσματα}
  \vspace{0.3cm}
  \begin{enumerate}
    \setlength{\itemsep}{0.55cm}
    \item Το \en{CascadeLP} συνδυάζει δομή και κείμενο, ξεπερνά κάθε

      λύσεις του \en{DSAA} με ελαφριά μοντέλα σε \en{CPU}.
    \item Οι σχεδόν τέλειες βαθμολογίες του \en{DSAA 2023} οφείλονται στη

      κείμενο και το πλεονέκτημά της μεγαλώνει· με μικτά αρνητικά παραμένει
      πρώτη και στα δύσκολα ζεύγη.

\begin{frame}{Επιπλέον έλεγχοι \en{DSAA}}

    \item \textbf{\en{SVM}:} εκπαίδευση σε \SvmSampleSize{} ζεύγη λόγω
      κόστους· όχι σύγκριση ίσου μεγέθους.
  \end{itemize}
\end{frame}

\begin{frame}{Όρια σύγκρισης και εμπιστοσύνης}
  \begin{itemize}
    \setlength{\itemsep}{0.5cm}
    \item \textbf{Όρια δρομολόγησης:} \TauOneDefault{} και
      \TauTwoDefault{} είναι χειρωνακτικές επιλογές. Οι πιθανότητες
      χρησιμοποιούνται ως βαθμολογίες, χωρίς ξεχωριστή βαθμονόμηση.
    \item \textbf{Διαστήματα:} ζευγαρωμένο \en{bootstrap}
      (\WfBootstrapResamples{} επαναδειγματοληψίες) σε σταθερά ζεύγη και
      μοντέλα· δεν καλύπτει αλλαγή σπόρου, διαίρεσης ή επανεκπαίδευση.
    \item \textbf{Κόστος:} οι ενσωματώσεις είχαν προϋπολογιστεί· η
      δρομολόγηση δείχνει πού αποφασίζεται κάθε ζεύγος, όχι μετρημένη
      εξοικονόμηση χρόνου.
    \item \textbf{\en{Cold-start}:} μόνο \WfColdStartCount{} λειτουργικά
      ζεύγη στο νέο σύνολο· δεν στηρίζουν ξεχωριστό συμπέρασμα.
  \end{itemize}
\end{frame}
